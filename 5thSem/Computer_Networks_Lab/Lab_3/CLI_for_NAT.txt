#Configure NAT on Local Router
interface g0/0
 ip nat inside

interface g1/0
 ip nat inside

interface g2/0
 ip nat inside

interface g3/0
 ip nat outside

#Static NAT for PC0 & PC1
ip nat inside source static 192.168.1.10 110.35.36.10
ip nat inside source static 192.168.1.11 110.35.36.11

#Dynamic NAT for PC2 & PC3
access-list 1 deny host 192.168.1.10
access-list 1 deny host 192.168.1.11
access-list 1 permit 192.168.2.0 0.0.0.255
ip nat pool DYN_POOL 110.35.36.12 110.35.36.13 netmask 255.255.255.0
ip nat inside source list 1 pool DYN_POOL

#PAT (Overload) for PC4 & PC5
access-list 2 permit 192.168.3.0 0.0.0.255
ip nat inside source list 2 interface g3/0 overload
